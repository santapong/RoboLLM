# Version constraints that MUST hold for ROS 2 Jazzy to keep working in the
# /opt/mpvenv venv. ROS 2 Jazzy is built against numpy 1.26.x; numpy 2.x
# breaks rclpy message conversions and cv_bridge. Always install with this
# constraints file so a transitive dep (mediapipe, pinocchio, pink, ...)
# can't silently drag numpy 2.x in through the back door:
#
#     /opt/mpvenv/bin/pip install -c constraints.txt <anything>
#
# Mirrors ../../../constraints.txt (repo root) — kept as a local copy so the
# docker build context (this docker/ dir) is self-contained.
numpy==1.26.4

# opencv-contrib-python 5.x hard-requires numpy>=2, so an unpinned install
# drags numpy 2.x in through the back door even when numpy itself is pinned
# above. Symptom when it slips: cv_bridge raises `KeyError: 16` at first use,
# because its C extension was compiled against numpy 1.x.
opencv-contrib-python<5
